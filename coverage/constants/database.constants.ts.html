export const CREATE_GAMES_TABLE = `
  CREATE TABLE IF NOT EXISTS games (
    game_id INTEGER PRIMARY KEY,
    white_player TEXT NOT NULL,
    black_player TEXT NOT NULL,
    winner TEXT NOT NULL,
    tournament TEXT,
    pgn TEXT NOT NULL,
    UNIQUE(pgn)
  );
`;
